def _page_of(doc, sentence: str) -> int:
    for page in doc.pages:
        if sentence in page.text:
            return page.page_number
    best, best_sc = 1, -1
    words = sentence.split()
    for page in doc.pages:
        n = sum(1 for w in words if w in page.text)
        if n > best_sc:
            best, best_sc = page.page_number, n
    return best

    """Yield (page_number, sentence_text) for every page of a document."""

def _join_paragraph_breaks(text: str) -> str:
    """Re-join lines that were broken mid-paragraph in the source.
    We must not join across a real blank line: hard line breaks (blank line)
    are trusted as paragraph boundaries in the fixture corpus.  Soft breaks
    (paragraph continues on the very next line) are joined so multi-word
    quoted snippets in DOCX data pieces survive intact.
    """
    lines = text.splitlines()
    joined: list[str] = []
    for line in lines:
        s = line.strip()
        if not s:  # blank line -> paragraph boundary
            joined.append("")
            continue
        if len(joined) > 0 and joined[-1] and not _ends_sentenceish(joined[-1]):
            joined[-1] += " " + s
        else:
            joined.append(s)
    return "\n".join(joined)

def iter_sentences(doc: Document):
    """Yield (page_number, sentence_text) for every page of a document."""
    for page in doc.pages:
        for paragraph in re.split(r"\n\s*\n", page.text):
            for sentence in _SENTENCE_SPLIT.split(paragraph):
                sentence = sentence.strip()
                if not sentence:
                    continue
                if not re.search(r"[0-9A-Za-z\u4e00-\u9fff]", sentence):
                    continue
                if re.fullmatch(r"[0-9A-Za-z.]{1,4}", sentence):
                    continue  # bare numbering / fragments
                yield page.page_number, sentence

def document_risk(clauses: list[ExtractedClause]) -> RiskLevel:
    """Aggregate per-document risk from its clauses."""
    if not clauses:
        return RiskLevel.LOW
    if any(c.risk == RiskLevel.HIGH for c in clauses):
        return RiskLevel.HIGH
    if any(c.risk == RiskLevel.MEDIUM for c in clauses):
        return RiskLevel.MEDIUM
    return RiskLevel.LOW

def _page_of(doc, sentence: str) -> int:
    for page in doc.pages:
        if sentence in page.text:
            return page.page_number
    best, best_sc = 1, -1
    words = sentence.split()
    for page in doc.pages:
        n = sum(1 for w in words if w in page.text)
        if n > best_sc:
            best, best_sc = page.page_number, n
    return best

def extract_document(
    path: str,
    source_format: str | None = None,
    use_llm: bool = False,
    title: str | None = None,
    progress: callable | None = None,
) -> DocumentResult:

def _join_paragraph_breaks(text: str) -> str:
    """Re-join lines that were broken mid-paragraph in the source.
    We must not join across a real blank line: hard line breaks (blank line)
    are trusted as paragraph boundaries in the fixture corpus.  Soft breaks
    (paragraph continues on the very next line) are joined so multi-word
    quoted snippets in DOCX data pieces survive intact.
    """
    lines = text.splitlines()
    joined: list[str] = []
    for line in lines:
        s = line.strip()
        if not s:  # blank line -> paragraph boundary
            joined.append("")
            continue
        if len(joined) > 0 and joined[-1] and not _ends_sentenceish(joined[-1]):
            joined[-1] += " " + s
        else:
            joined.append(s)
    return "\n".join(joined)
_SENTENCE_END = re.compile(r"[.!?:。！？；;」』）)]\s*$")
def _ends_sentenceish(s: str) -> bool:
    return bool(_SENTENCE_END.search(s))
def _normalise_text(doc_text: str) -> str:
    """Stage 1: DOCX/PdfPlumber text clean-up.
    - hyphenation across line breaks,
    - collapse multiple spaces,
    - re-join soft line breaks,
    - normalise CRLF.
    """
    text = doc_text.replace("\r\n", "\n").replace("\r", "\n")
    text = _restore_hyphenation(text)
    text = re.sub(r"[ \t]+", " ", text)
    text = _join_paragraph_breaks(text)
    text = re.sub(r"[ \t]+\n", "\n", text)
    text = re.sub(r"\n{3,}", "\n\n", text)
    return text.strip()

def iter_documents(paths: list[Path]) -> list[Path]:
    """Expand directories into sorted pdf/docx file paths."""
    out: list[Path] = []

def _write_checklists_from_ndjson(path: str) -> None:
    ndjson = Path(path)

            data = json.loads(line)

def _all_risk(groups: list[RiskLevel]) -> RiskLevel:
    if any(r == RiskLevel.HIGH for r in groups):
        return RiskLevel.HIGH
    if any(r == RiskLevel.MEDIUM for r in groups):
        return RiskLevel.MEDIUM
    return RiskLevel.LOW
